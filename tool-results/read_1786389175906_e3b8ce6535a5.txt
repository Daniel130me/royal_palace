   100→function SoapField({ label, value, onChange, disabled, placeholder, required, error }: {
   101→  label: string; value: string | undefined; onChange: (v: string) => void; disabled: boolean; placeholder?: string; required?: boolean; error?: boolean;
   102→}) {
   103→  return (
   104→    <div>
   105→      <Label className="text-xs font-medium text-muted-foreground">
   106→        {label} {required && <span className="text-rose-500">*</span>}
   107→      </Label>
   108→      <Textarea
   109→        rows={2}
   110→        value={value ?? ""}
   111→        onChange={(e) => onChange(e.target.value)}
   112→        disabled={disabled}
   113→        placeholder={placeholder}
   114→        aria-invalid={error || undefined}
   115→        className={`mt-1 ${error ? "ring-2 ring-rose-400/60 border-rose-300" : ""}`}
   116→      />
   117→      {error && <p className="mt-1 text-[11px] text-rose-600">This field is required.</p>}
   118→    </div>
   119→  );
   120→}

   122→export function ProviderEncounter() {
   123→  const { view } = useNav();
   124→  const id = view.params.id;
   125→  const [encounter, setEncounter] = useState<ClinicalEncounter | null>(null);
   126→  const [patient, setPatient] = useState<Patient | null>(null);
   127→  const [history, setHistory] = useState<Appointment[]>([]);
   128→  const [prescriptions, setPrescriptions] = useState<Prescription[]>([]);
   129→  const [labRequests, setLabRequests] = useState<LaboratoryRequest[]>([]);
   130→  const [loading, setLoading] = useState(true);
   131→  const [error, setError] = useState<string | null>(null);
   132→  const [doc, setDoc] = useState<EncounterDocumentation>({});
   133→  const [diagnosisCode, setDiagnosisCode] = useState("");
   134→  const [attachments, setAttachments] = useState<FileMeta[]>([]);
   135→  const [saving, setSaving] = useState<"idle" | "saving" | "saved">("idle");
   136→  const [completing, setCompleting] = useState(false);
   137→  const [missingFields, setMissingFields] = useState<string[]>([]);
   138→  const [rxOpen, setRxOpen] = useState(false);
   139→  const [labOpen, setLabOpen] = useState(false);
   140→  const [refOpen, setRefOpen] = useState(false);
   141→  /** Mobile collapsible for patient summary */
   142→  const [summaryOpen, setSummaryOpen] = useState(false);
   143→
   144→  // autosave bookkeeping
   145→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   146→  const lastSavedRef = useRef<string>("");
   147→  const providerId = useNav((s) => s.session?.profileId ?? "");
   148→
   149→  const load = useCallback(async () => {
   150→    if (!id) return;
   151→    setLoading(true);
   152→    setError(null);
   153→    try {
   154→      const enc = await encounterService.get(id);
   155→      const norm = normalizeEncounter(enc);
   156→      setEncounter(norm);
   157→      const initialDoc = norm.documentation;
   158→      setDoc(initialDoc);
   159→      setDiagnosisCode(initialDoc.diagnosis ? (initialDoc as EncounterDocumentation & { diagnosisCode?: string }).diagnosisCode ?? "" : "");
   160→      setAttachments(initialDoc.attachments ?? []);
   161→      lastSavedRef.current = JSON.stringify(initialDoc);
   162→
   163→      if (norm.patientId) {
   164→        const [p, appts, rxs, labs] = await Promise.all([
   165→          patientService.get(norm.patientId),
   166→          appointmentService.list({ patientId: norm.patientId }),
   167→          prescriptionService.list({ patientId: norm.patientId }),
   168→          labRequestService.list({ patientId: norm.patientId }),
   169→        ]);
   170→        setPatient(normalizePatient(p));
   171→        setHistory(appts);
   172→        setPrescriptions(rxs);
   173→        setLabRequests(labs.map(normalizeLabRequest));
   174→      }
   175→    } catch (e) {
   176→      setError((e as Error).message ?? "Failed to load encounter.");
   177→    } finally {
   178→      setLoading(false);
   179→    }
   180→  }, [id]);
   181→
   182→  useEffect(() => {
   183→    load();
   184→  }, [load]);
   185→
   186→  // Autosave: debounced 1.5s after last keystroke
   187→  const docJson = JSON.stringify(doc);
   188→  useEffect(() => {
   189→    if (!encounter || encounter.locked) return;
   190→    if (docJson === lastSavedRef.current) return;
   191→    setSaving("saving");
   192→    if (saveTimer.current) clearTimeout(saveTimer.current);
   193→    saveTimer.current = setTimeout(async () => {
   194→      try {
   195→        const payload: DocWithCode = { ...doc, diagnosisCode, attachments };
   196→        await encounterService.update(encounter.id, payload);
   197→        lastSavedRef.current = JSON.stringify(doc);
   198→        setSaving("saved");
   199→        setTimeout(() => setSaving("idle"), 1500);
   200→      } catch (e) {
   201→        toast.error("Autosave failed: " + (e as Error).message);
   202→        setSaving("idle");
   203→      }
   204→    }, 1500);
   205→    return () => {
   206→      if (saveTimer.current) clearTimeout(saveTimer.current);
   207→    };
   208→  }, [doc, diagnosisCode, attachments, encounter]);
   209→
   210→  function update<K extends keyof EncounterDocumentation>(key: K, value: EncounterDocumentation[K]) {
   211→    setDoc((prev) => ({ ...prev, [key]: value }));
   212→  }
   213→  function updateVital(key: string, value: string) {
   214→    setDoc((prev) => ({ ...prev, vitalSigns: { ...(prev.vitalSigns ?? {}), [key]: value } }));
   215→  }
   216→  function addMockAttachment() {
   217→    const names = ["EKG.pdf", "Patient-Photo.jpg", "Wound.jpg", "Lab-Requisition.pdf"];
   218→    const name = names[Math.floor(Math.random() * names.length)];
   219→    setAttachments((prev) => [
   220→      ...prev,
   221→      { id: `ATT-${Date.now().toString(36)}`, name, type: name.endsWith(".pdf") ? "application/pdf" : "image/jpeg", size: 100_000 + Math.floor(Math.random() * 200_000), uploadedAt: new Date().toISOString() },
   222→    ]);
   223→  }
   224→  function removeAttachment(idx: number) {
   225→    setAttachments((prev) => prev.filter((_, i) => i !== idx));
   226→  }
   227→
   228→  async function complete() {
   229→    if (!encounter) return;
   230→    const missing = REQUIRED_FIELDS
   231→      .filter((f) => !doc[f.key] || String(doc[f.key]).trim() === "")
   232→      .map((f) => f.key as string);
   233→    if (missing.length > 0) {
   234→      setMissingFields(missing);
   235→      toast.error(`Cannot complete: ${missing.length} required field(s) missing.`);
   236→      return;
   237→    }
   238→    setMissingFields([]);
   239→    setCompleting(true);
   240→    try {
   241→      const payload: DocWithCode = { ...doc, diagnosisCode, attachments };
   242→      await encounterService.complete(encounter.id, payload, providerId);
   243→      toast.success("Consultation completed and clinical record signed.");
   244→      await load();
   245→    } catch (e) {
   246→      toast.error("Could not complete encounter: " + (e as Error).message);
   247→    } finally {
   248→      setCompleting(false);
   249→    }
   250→  }
   251→
   252→  const locked = !!encounter?.locked;
   253→  const recentEncounters = useMemo(
   254→    () => history.filter((a) => a.encounter && a.encounter.id !== id).slice(0, 5),
   255→    [history, id]
   256→  );
   257→  const recentLabs = useMemo(
   258→    () => labRequests.filter((l) => l.result).slice(0, 5),
   259→    [labRequests]
   260→  );
   261→  const activeRx = useMemo(
   262→    () => prescriptions.filter((p) => ["issued", "awaiting_pharmacy", "partially_fulfilled"].includes(p.status)).slice(0, 5),
   263→    [prescriptions]
   264→  );
   265→
   266→  const isMissing = (key: keyof EncounterDocumentation) => missingFields.includes(key as string);
   267→
   268→  if (loading) return <LoadingState label="Loading encounter workspace…" />;
   269→  if (error) return <ErrorState message={error} onRetry={load} />;
   270→  if (!encounter) return <ErrorState message="Encounter not found." />;
   271→  if (!patient) return <ErrorState message="Patient record missing." />;
   272→
   273→  return (
   274→    <div className="pb-28 lg:pb-0">
   275→      <PageHeader
   276→        title={`Encounter ${encounter.encounterNumber}`}
   277→        description={`${patient.firstName} ${patient.lastName} · ${patient.patientNumber} · Started ${formatDate(encounter.createdAt)}`}
   278→        back
   279→        actions={
   280→          <div className="flex items-center gap-2">
   281→            <StatusBadge status={encounter.status} />
   282→            {locked ? (
   283→              <Badge variant="outline" className="border-rose-200 bg-rose-50 text-rose-700 h-6">
   284→                <Lock className="h-3 w-3 mr-1" /> Signed
   285→              </Badge>
   286→            ) : (
   287→              <Badge
   288→                variant="outline"
   289→                className={`h-6 ${
   290→                  saving === "saving"
   291→                    ? "border-amber-200 bg-amber-50 text-amber-700"
   292→                    : saving === "saved"
   293→                    ? "border-emerald-200 bg-emerald-50 text-emerald-700"
   294→                    : ""
   295→                }`}
   296→              >
   297→                {saving === "saving" ? (<><Save className="h-3 w-3 mr-1 animate-pulse" /> Saving…</>) :
   298→                 saving === "saved" ? (<><CheckCircle2 className="h-3 w-3 mr-1" /> Saved</>) :
   299→                 (<><Save className="h-3 w-3 mr-1" /> Autosave on</>)}
   300→              </Badge>
   301→            )}
   302→          </div>
   303→        }
   304→      />
   305→
   306→      {locked && (
   307→        <Alert className="mb-5 border-emerald-200 bg-emerald-50">
   308→          <ShieldCheck className="h-4 w-4 text-emerald-600" />
   309→          <AlertTitle className="text-emerald-800">Signed Clinical Record — This consultation is locked.</AlertTitle>
   310→          <AlertDescription className="text-emerald-700">
   311→            Signed on {encounter.signedAt ? formatDate(encounter.signedAt) : "—"} by {encounter.provider ? `${encounter.provider.title} ${encounter.provider.lastName}` : "the consulting provider"}. The documentation below is read-only.
   312→          </AlertDescription>
   313→        </Alert>
   314→      )}
   315→
   316→      {/* Mobile: collapsible patient summary banner */}
   317→      <div className="lg:hidden mb-4">
   318→        <Collapsible open={summaryOpen} onOpenChange={setSummaryOpen}>
   319→          <button
   320→            className="flex w-full items-center gap-3 rounded-2xl border border-border/80 bg-card p-3 shadow-soft tap-highlight-none"
   321→            onClick={() => setSummaryOpen((o) => !o)}
   322→          >
   323→            <Avatar className="h-10 w-10">
   324→              <AvatarFallback className="bg-primary/10 text-primary text-xs font-semibold">
   325→                {patient.firstName[0]}{patient.lastName[0]}
   326→              </AvatarFallback>
   327→            </Avatar>
   328→            <div className="min-w-0 flex-1 text-left">
   329→              <p className="text-sm font-semibold truncate">{patient.firstName} {patient.lastName}</p>
   330→              <p className="text-xs text-muted-foreground truncate">
   331→                {age(patient.dateOfBirth) ?? "—"}y · {patient.gender} · {patient.bloodGroup ?? "?"}
   332→              </p>
   333→            </div>
   334→            {patient.allergies.filter((a) => a.status === "active").length > 0 && (
   335→              <Badge variant="outline" className="border-rose-200 bg-rose-50 text-rose-700 text-[10px]">
   336→                <AlertCircle className="h-3 w-3 mr-1" /> {patient.allergies.filter((a) => a.status === "active").length} allergy
   337→              </Badge>
   338→            )}
   339→            <ChevronDown className={`h-4 w-4 text-muted-foreground transition-transform ${summaryOpen ? "rotate-180" : ""}`} />
   340→          </button>
   341→          <CollapsibleContent>
   342→            <div className="mt-2 rounded-2xl border border-border/80 bg-card shadow-soft overflow-hidden">
   343→              <PatientSummaryBody
   344→                patient={patient}
   345→                recentEncounters={recentEncounters}
   346→                recentLabs={recentLabs}
   347→                activeRx={activeRx}
   348→              />
   349→            </div>
   350→          </CollapsibleContent>
   351→        </Collapsible>
   352→      </div>
   353→
   354→      {/* 3-panel workspace on desktop, stacked on mobile */}
   355→      <div className="grid gap-5 lg:grid-cols-[280px_1fr_320px] lg:items-start">
   356→        {/* LEFT: Patient summary (desktop sticky, mobile collapsed above) */}
   357→        <aside className="hidden lg:block lg:sticky lg:top-20">
   358→          <SectionCard
   359→            title="Patient summary"
   360→            icon={User}
   361→            dense
   362→            contentClassName="p-0"
   363→          >
   364→            <PatientSummaryBody
   365→              patient={patient}
   366→              recentEncounters={recentEncounters}
   367→              recentLabs={recentLabs}
   368→              activeRx={activeRx}
   369→            />
   370→          </SectionCard>
   371→        </aside>
   372→
   373→        {/* CENTER: SOAP documentation */}
   374→        <div className={locked ? "opacity-80" : ""}>
   375→          <SectionCard
   376→            title="Clinical documentation"
   377→            description="SOAP notes — autosaved as you type"
   378→            icon={FileText}
   379→            action={
   380→              !locked && (
   381→                <Badge
   382→                  variant="outline"
   383→                  className={`h-6 ${saving === "saving" ? "border-amber-200 bg-amber-50 text-amber-700" : saving === "saved" ? "border-emerald-200 bg-emerald-50 text-emerald-700" : ""}`}
   384→                >
   385→                  {saving === "saving" ? (<><Save className="h-3 w-3 mr-1 animate-pulse" /> Saving…</>) :
   386→                   saving === "saved" ? (<><CheckCircle2 className="h-3 w-3 mr-1" /> Saved</>) :
   387→                   (<><Save className="h-3 w-3 mr-1" /> Autosave on</>)}
   388→                </Badge>
   389→              )
   390→            }
   391→          >
   392→            <div className="space-y-4">
   393→              {missingFields.length > 0 && (
   394→                <Alert className="border-rose-200 bg-rose-50">
   395→                  <AlertCircle className="h-4 w-4 text-rose-600" />
   396→                  <AlertTitle className="text-rose-800">Required fields missing</AlertTitle>
   397→                  <AlertDescription className="text-rose-700">
   398→                    <ul className="list-disc ml-4 text-xs space-y-0.5">
   399→                      {missingFields.map((m) => {
   400→                        const f = REQUIRED_FIELDS.find((r) => r.key === m);
   401→                        return <li key={m}>{f?.label ?? m}</li>;
   402→                      })}
   403→                    </ul>
   404→                  </AlertDescription>
   405→                </Alert>
   406→              )}
   407→
   408→              <SoapField label="Consultation Reason" required error={isMissing("consultationReason")} value={doc.consultationReason} onChange={(v) => update("consultationReason", v)} disabled={locked} placeholder="Why the patient is being seen today." />
   409→              <SoapField label="Chief Complaint" value={doc.chiefComplaint} onChange={(v) => update("chiefComplaint", v)} disabled={locked} placeholder="In the patient's own words." />
   410→              <SoapField label="History of Present Illness" value={doc.historyPresentIllness} onChange={(v) => update("historyPresentIllness", v)} disabled={locked} placeholder="Onset, course, severity, modifying factors." />
   411→              <SoapField label="Relevant Medical History" required error={isMissing("relevantMedicalHistory")} value={doc.relevantMedicalHistory} onChange={(v) => update("relevantMedicalHistory", v)} disabled={locked} placeholder="Past conditions, hospitalisations, surgeries." />
   412→              <SoapField label="Medication History" required error={isMissing("medicationHistory")} value={doc.medicationHistory} onChange={(v) => update("medicationHistory", v)} disabled={locked} placeholder="Current and recent medications, adherence." />
   413→              <SoapField label="Allergy Confirmation" required error={isMissing("allergyConfirmation")} value={doc.allergyConfirmation} onChange={(v) => update("allergyConfirmation", v)} disabled={locked} placeholder="Confirm allergies reviewed with patient (or 'No known allergies')." />
   414→              <div className="grid gap-3 sm:grid-cols-2">
   415→                <SoapField label="Family History" value={doc.familyHistory} onChange={(v) => update("familyHistory", v)} disabled={locked} placeholder="Hereditary conditions." />
   416→                <SoapField label="Social History" value={doc.socialHistory} onChange={(v) => update("socialHistory", v)} disabled={locked} placeholder="Occupation, smoking, alcohol, lifestyle." />
   417→              </div>
   418→
   419→              <Separator />
   420→
   421→              {/* Vital signs grid */}
   422→              <div>
   423→                <Label className="text-xs font-medium text-muted-foreground">Vital Signs</Label>
   424→                <div className="grid gap-3 sm:grid-cols-3 mt-1.5">
   425→                  {VITAL_KEYS.map((v) => (
   426→                    <div key={v.key}>
   427→                      <Label className="text-[10px] text-muted-foreground uppercase tracking-wider">{v.label}</Label>
   428→                      <Input
   429→                        value={(doc.vitalSigns ?? {})[v.key] ?? ""}
   430→                        onChange={(e) => updateVital(v.key, e.target.value)}
   431→                        disabled={locked}
   432→                        placeholder={v.placeholder}
   433→                      />
   434→                    </div>
   435→                  ))}
   436→                </div>
   437→              </div>
   438→
   439→              <SoapField label="Examination Findings" value={doc.examinationFindings} onChange={(v) => update("examinationFindings", v)} disabled={locked} placeholder="System-by-system examination." />
   440→
   441→              <Separator />
   442→
   443→              <SoapField label="Assessment" required error={isMissing("assessment")} value={doc.assessment} onChange={(v) => update("assessment", v)} disabled={locked} placeholder="Clinical impression and reasoning." />
   444→              <div className="grid gap-3 sm:grid-cols-[2fr_1fr]">
   445→                <div>
   446→                  <Label className="text-xs font-medium text-muted-foreground">
   447→                    Diagnosis <span className="text-rose-500">*</span>
   448→                  </Label>
   449→                  <Input
   450→                    value={doc.diagnosis ?? ""}
   451→                    onChange={(e) => update("diagnosis", e.target.value)}
   452→                    disabled={locked}
   453→                    placeholder="e.g. Essential hypertension"
   454→                    aria-invalid={isMissing("diagnosis") || undefined}
   455→                    className={`mt-1 ${isMissing("diagnosis") ? "ring-2 ring-rose-400/60 border-rose-300" : ""}`}
   456→                  />
   457→                  {isMissing("diagnosis") && <p className="mt-1 text-[11px] text-rose-600">This field is required.</p>}
   458→                </div>
   459→                <div>
   460→                  <Label className="text-xs font-medium text-muted-foreground">Provisional code</Label>
   461→                  <Input value={diagnosisCode} onChange={(e) => setDiagnosisCode(e.target.value)} disabled={locked} placeholder="e.g. I10 (ICD-10)" className="mt-1" />
   462→                </div>
   463→              </div>
   464→              <SoapField label="Differential Diagnosis" value={doc.differentialDiagnosis} onChange={(v) => update("differentialDiagnosis", v)} disabled={locked} placeholder="Alternative diagnoses considered." />
